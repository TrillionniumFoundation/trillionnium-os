def need(value,message):
    if not value:raise ValueError(message)


def pin(raw):
    need(type(raw)is bytes,'complete original bytes required')
    return {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}


def strict(raw):
    need(type(raw)is bytes and len(raw)<=MAX_ORIGINAL,'bounded complete original JSON')
    def pairs(items):
        value={}
        for key,item in items:
            need(key not in value,'duplicate original JSON key');value[key]=item
        return value
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=lambda _:need(False,'nonfinite JSON'))


def only_function(tree,name):
    rows=[n for n in tree.body if isinstance(n,ast.FunctionDef)and n.name==name]
    need(len(rows)==1,'one exact source function: '+name)
    return rows[0]


def exact_ast(node, spelling, *, statement=False):
    """Compare complete AST fields, never interpreter-specific pretty printing."""
    parsed = ast.parse(spelling, mode='exec' if statement else 'eval')
    if statement:
        need(len(parsed.body) == 1, 'one exact reference statement')
        expected = parsed.body[0]
    else:
        expected = parsed.body
    return ast.dump(node, include_attributes=False) == ast.dump(expected, include_attributes=False)


def only_call(function,name,*,line=None):
    rows=[n for n in ast.walk(function)if isinstance(n,ast.Call)
          and exact_ast(n.func,name)and(line is None or n.lineno==line)]
    need(len(rows)==1,'one exact source call: '+name)
    return rows[0]
