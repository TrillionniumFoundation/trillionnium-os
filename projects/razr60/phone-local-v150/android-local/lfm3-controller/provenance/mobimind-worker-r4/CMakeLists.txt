cmake_minimum_required(VERSION 3.22)
project(hepta_mobimind_model_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This worker requires Android arm64-v8a")
endif()
if(NOT ANDROID_STL STREQUAL "c++_shared")
  message(FATAL_ERROR "Use the same NDK c++_shared runtime as libMNN")
endif()
if(NOT EXISTS "${MNN_SOURCE}/transformers/llm/engine/include/llm/llm.hpp" OR NOT EXISTS "${MNN_LIBRARY}")
  message(FATAL_ERROR "Set MNN_SOURCE to the pinned source and MNN_LIBRARY to the real built libMNN.so")
endif()
add_library(qualified_mnn SHARED IMPORTED)
set_target_properties(qualified_mnn PROPERTIES IMPORTED_LOCATION "${MNN_LIBRARY}")
add_executable(hepta-mobimind-probe native/worker.cpp)
target_include_directories(hepta-mobimind-probe PRIVATE
  "${MNN_SOURCE}/include" "${MNN_SOURCE}/transformers/llm/engine/include"
  "${MNN_SOURCE}/tools/cv/include")
target_compile_options(hepta-mobimind-probe PRIVATE -Wall -Wextra -Wpedantic -fPIE)
target_link_options(hepta-mobimind-probe PRIVATE -pie -Wl,--export-dynamic -Wl,-z,relro,-z,now -Wl,-rpath,\$ORIGIN)
target_link_libraries(hepta-mobimind-probe PRIVATE qualified_mnn log android dl)
